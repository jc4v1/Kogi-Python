\begin{algorithm}[H]
\caption{fire\_elements}
\begin{algorithmic}[1]
\REQUIRE Set of elements $E$
\ENSURE Fires all elements in $E$ according to rules and propagates changes to parents
\FOR{each $e$ in $E$}
    \IF{try\_any\_rule($e$) succeeds}
        \STATE add $e$ to changed elements
        \STATE $P \gets$ parents of $e$
        \STATE fire\_elements($P$)
    \ENDIF
\ENDFOR
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_any\_rule}
\begin{algorithmic}[1]
\REQUIRE Element $e$
\ENSURE Returns True if any rule applies to $e$, False otherwise
\IF{try\_pie\_rule($e$) succeeds}
    \RETURN True
\ELSIF{try\_por\_rule($e$) succeeds}
    \RETURN True
\ELSIF{try\_pand\_rule($e$) succeeds}
    \RETURN True
\ELSIF{try\_pmake\_rule($e$) succeeds}
    \RETURN True
\ELSIF{try\_pbreak\_rule($e$) succeeds}
    \RETURN True
\ELSIF{try\_bpfulfill\_rule($e$) succeeds}
    \RETURN True
\ELSIF{try\_bpdeny\_rule($e$) succeeds}
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_pie\_rule}
\begin{algorithmic}[1]
\REQUIRE Element $e$
\ENSURE Returns True if PIE rule applied, False otherwise
\IF{$e$ exists and is a leaf and status$(e) \neq (\top, \bot)$}
    \STATE set status$(e) \gets (\top, \bot)$
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_pand\_rule}
\begin{algorithmic}[1]
\REQUIRE Element $e$
\ENSURE Returns True if PAND rule applied, False otherwise
\STATE $and\_links \gets$ all AND links from $e$
\IF{$and\_links$ not empty and all children have status $(\top, \bot)$}
    \STATE set status$(e) \gets (\top, \bot)$
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_por\_rule}
\begin{algorithmic}[1]
\REQUIRE Element $e$
\ENSURE Returns True if POR rule applied, False otherwise
\STATE $or\_links \gets$ all OR links from $e$
\IF{any child of $or\_links$ has status $(\top, \bot)$}
    \STATE set status$(e) \gets (\top, \bot)$
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_pmake\_rule}
\begin{algorithmic}[1]
\REQUIRE Quality $q$
\ENSURE Returns True if PMAKE rule applied, False otherwise
\STATE $make\_links \gets$ all MAKE links from $q$
\IF{any source of $make\_links$ has status $(\top, \bot)$ and status$(q) = ?$}
    \STATE set status$(q) \gets \top$
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_pbreak\_rule}
\begin{algorithmic}[1]
\REQUIRE Quality $q$
\ENSURE Returns True if PBREAK rule applied, False otherwise
\STATE $break\_links \gets$ all BREAK links from $q$
\IF{any source of $break\_links$ has status $(\top, \bot)$ and status$(q) = ?$}
    \STATE set status$(q) \gets \bot$
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_bpfulfill\_rule}
\begin{algorithmic}[1]
\REQUIRE Quality $q$
\ENSURE Returns True if BPFULFILL rule applied, False otherwise
\STATE $make\_links \gets$ all MAKE links from $q$
\IF{any source of $make\_links$ has status $(\top, \bot)$ and status$(q) = \bot$}
    \STATE set status$(q) \gets \top$
    \FOR{each BREAK link $b$ from $q$ with source status $(\top, \bot)$}
        \FOR{each refinement $e$ of $b$}
            \IF{status$(e) = (\top, \bot)$}
                \STATE set status$(e) \gets (\top, \top)$
            \ENDIF
        \ENDFOR
    \ENDFOR
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{try\_bpdeny\_rule}
\begin{algorithmic}[1]
\REQUIRE Quality $q$
\ENSURE Returns True if BPDENY rule applied, False otherwise
\STATE $break\_links \gets$ all BREAK links from $q$
\IF{any source of $break\_links$ has status $(\top, \bot)$ and status$(q) = \top$}
    \STATE set status$(q) \gets \bot$
    \FOR{each MAKE link $m$ from $q$ with source status $(\top, \bot)$}
        \FOR{each refinement $e$ of $m$}
            \IF{status$(e) = (\top, \bot)$}
                \STATE set status$(e) \gets (\top, \top)$
            \ENDIF
        \ENDFOR
    \ENDFOR
    \RETURN True
\ELSE
    \RETURN False
\ENDIF
\end{algorithmic}
\end{algorithm}
